How to train locally

python3 -m venv .venv && source .venv/bin/activate
pip install -r requirements.txt

# 1. Corpus 85M chars → 20M tokens
python -m llm.build_corpus --out data/corpus.txt --chars 85000000

# 2. Tokenizer 4096 vocab
python -m llm.tokenizer --corpus data/corpus.txt --out checkpoints/tokenizer.json

# 3. Train (2442 steps)
python -m llm.train

# 4. Generate
python -m llm.generate --prompt "Once upon a time, Lily" --tokens 150
